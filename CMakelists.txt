cmake_minimum_required(VERSION 3.16)

project(NetworkHealthMonitor LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)


#tcp server setup
add_executable(network_health_monitor_server
    src/server/tcp_server.cpp
    src/server/server_main.cpp
    src/database/database_manager.cpp
)

target_include_directories(network_health_monitor_server PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/Include
    ${CMAKE_CURRENT_SOURCE_DIR}
)

target_link_libraries(network_health_monitor_server PRIVATE ws2_32)

#tcp server client setup 
add_executable(network_health_monitor_client
    src/client/client_sim.cpp
    src/client/client_main.cpp
    src/client/client_telemetry.cpp
)

target_include_directories(network_health_monitor_client PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/Include
    ${CMAKE_CURRENT_SOURCE_DIR}
)

target_link_libraries(network_health_monitor_client PRIVATE ws2_32)
