cmake_minimum_required(VERSION 3.16)

project(NetworkHealthMonitor LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(FetchContent)

FetchContent_Declare(
    Crow
    GIT_REPOSITORY https://github.com/CrowCpp/Crow.git
    GIT_TAG master
)

FetchContent_MakeAvailable(Crow)

# Standalone Phase 4A REST server.
add_executable(network_health_monitor_rest
    src/main.cpp
    src/api/rest_server.cpp
    src/api/system_routes.cpp
    src/database/database_manager.cpp
    src/database/database_toolkit.cpp
)

target_include_directories(network_health_monitor_rest PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/Include
    "C:/msys64/ucrt64/include"
    "D:/postgreSql/include"
)

target_link_directories(network_health_monitor_rest PRIVATE
    "C:/msys64/ucrt64/lib"
    "D:/postgreSql/lib"
)

target_link_libraries(network_health_monitor_rest PRIVATE
    Crow::Crow
    ws2_32
    mswsock
    pq
)

#tcp server setup
add_executable(network_health_monitor_server
    src/server/tcp_server.cpp
    src/server/server_main.cpp
    src/database/database_manager.cpp
    src/database/database_toolkit.cpp
)

target_include_directories(network_health_monitor_server PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/Include
    ${CMAKE_CURRENT_SOURCE_DIR}
)

target_link_libraries(network_health_monitor_server PRIVATE ws2_32)

#tcp server client setup 
add_executable(network_health_monitor_client
    src/client/client_sim.cpp
    src/client/client_main.cpp
    src/client/client_telemetry.cpp
)

target_include_directories(network_health_monitor_client PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/Include
    ${CMAKE_CURRENT_SOURCE_DIR}
)

target_link_libraries(network_health_monitor_client PRIVATE ws2_32)

# PostgreSQL
target_include_directories(network_health_monitor_server PRIVATE
    "C:/msys64/ucrt64/include"
)

target_link_directories(network_health_monitor_server PRIVATE
    "C:/msys64/ucrt64/lib"
)

target_link_libraries(network_health_monitor_server PRIVATE
    ws2_32
    pq
)
